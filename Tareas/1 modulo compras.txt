   Referencia de la lógica (adáptala a los nombres reales del schema):

```python
   from decimal import Decimal, ROUND_HALF_UP
   from database.db import obtener_conexion
   from services.unidades import convertir

   class ErrorCompra(ValueError):
       """Error de validación con mensaje en español para mostrar al usuario."""

   def _q(valor, decimales="0.0001"):
       return Decimal(valor).quantize(Decimal(decimales), rounding=ROUND_HALF_UP)

   def registrar_compra(proveedor_id, fecha, numero_documento, observaciones, lineas):
       if not lineas:
           raise ErrorCompra("Agrega al menos un producto a la compra.")
       con = obtener_conexion()
       try:
           cur = con.execute(
               "INSERT INTO compras (proveedor_id, fecha, numero_documento, observaciones, total)"
               " VALUES (?, ?, ?, ?, 0)",
               (proveedor_id, fecha, numero_documento, observaciones))
           compra_id = cur.lastrowid
           total = Decimal("0")
           for l in lineas:                       # l: producto_id, cantidad, unidad_id, precio_total (Decimal)
               if l["cantidad"] <= 0 or l["precio_total"] <= 0:
                   raise ErrorCompra("Cantidad y precio deben ser mayores a cero.")
               prod = con.execute("SELECT * FROM productos WHERE id = ?", (l["producto_id"],)).fetchone()
               u_linea = con.execute("SELECT * FROM unidades WHERE id = ?", (l["unidad_id"],)).fetchone()
               u_compra = con.execute("SELECT * FROM unidades WHERE id = ?", (prod["unidad_compra_id"],)).fetchone()
               try:
                   cant_en_u_compra = convertir(l["cantidad"], u_linea, u_compra)
               except ValueError:
                   raise ErrorCompra("La unidad elegida no es compatible con la del producto.")
               precio_unit = _q(l["precio_total"] / cant_en_u_compra)   # $ por unidad de COMPRA del producto
               det = con.execute(
                   "INSERT INTO detalle_compras (compra_id, producto_id, cantidad, unidad_id, precio_total, precio_unitario)"
                   " VALUES (?, ?, ?, ?, ?, ?)",
                   (compra_id, prod["id"], float(l["cantidad"]), u_linea["id"],
                    float(l["precio_total"]), float(precio_unit)))
               ultimo = con.execute(
                   "SELECT MAX(fecha) FROM historial_precios WHERE producto_id = ?", (prod["id"],)).fetchone()[0]
               con.execute(
                   "INSERT INTO historial_precios (producto_id, fecha, precio_unitario, unidad_id, detalle_compra_id)"
                   " VALUES (?, ?, ?, ?, ?)",
                   (prod["id"], fecha, float(precio_unit), prod["unidad_compra_id"], det.lastrowid))
               # Una factura vieja cargada hoy NO debe pisar un precio más reciente:
               if ultimo is None or fecha >= ultimo:
                   con.execute("UPDATE productos SET precio_actual = ? WHERE id = ?",
                               (float(precio_unit), prod["id"]))
               total += l["precio_total"]
           con.execute("UPDATE compras SET total = ? WHERE id = ?", (float(total), compra_id))
           con.commit()
           return compra_id
       except Exception:
           con.rollback()
           raise
       finally:
           con.close()
```
